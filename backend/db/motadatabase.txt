package "Module: Authentication & User Management" {
  entity "users [CO SAN]" as users <<auth>> {
    * id : integer <<PK>>
    --
    username : varchar
    hashed_password : varchar
    full_name : varchar
    avatar : varchar
    created_at : timestamptz
    ..bo sung..
    + email : varchar <<UNIQUE>>
    + bio : text
    + is_private : boolean
    + role : varchar
    + status : varchar
    + updated_at : timestamptz
  }

  entity "user_tokens" as user_tokens <<auth>> {
    * id : integer <<PK>>
    --
    # user_id : integer <<FK>>
    token : varchar <<UNIQUE>>
    type : varchar
    expires_at : timestamptz
    created_at : timestamptz
  }
}

' ---------------------------------------------------------------------
package "Module: Recipe Management" {
  entity "recipes [CO SAN]" as recipes <<recipe>> {
    * id : integer <<PK>>
    --
    # user_id : integer <<FK>>
    recipe_name : varchar
    ingredients : json
    instructions : json
    created_at : timestamptz
    image_url : varchar
    ..bo sung..
    + description : text
    + cook_time_minutes : integer
    + servings : integer
    + difficulty : varchar
    + updated_at : timestamptz
  }

  entity "categories" as categories <<recipe>> {
    * id : integer <<PK>>
    --
    name : varchar <<UNIQUE>>
    slug : varchar <<UNIQUE>>
    created_at : timestamptz
  }

  entity "recipe_categories" as recipe_categories <<recipe>> {
    * id : integer <<PK>>
    --
    # recipe_id : integer <<FK>>
    # category_id : integer <<FK>>
    created_at : timestamptz
  }
}

' ---------------------------------------------------------------------

    # recipe_id : integer <<FK>>
    ingredient_name : varchar
  }
}

' ---------------------------------------------------------------------
package "Module: AI Ingredient Detection" {
  entity "ai_detection_histories" as ai_detection_histories <<ai>> {
    * id : integer <<PK>>
    --
    # user_id : integer <<FK>>
    image_url : varchar
    detected_ingredients : jsonb
    created_at : timestamptz
  }

  entity "recipe_stats [CO SAN]" as recipe_stats <<ai>> {
    * id : integer <<PK>>
    --
    recipe_name : varchar <<UNIQUE>>
    scan_count : integer
    save_count : integer
    ingredients : json
    instructions : json
    image_url : varchar
    created_at : timestamptz
    updated_at : timestamptz
    ..bo sung..
    + recipe_id : integer <<FK, NULL>>
  }
}

' ---------------------------------------------------------------------
package "Module: Community Interaction" {
  entity "comments" as comments <<community>> {
    * id : integer <<PK>>
    --
    # recipe_id : integer <<FK>>
    # user_id : integer <<FK>>
    # parent_comment_id : integer <<FK, NULL>>
    content : text
    created_at : timestamptz
    updated_at : timestamptz
  }

  entity "comment_likes" as comment_likes <<community>> {
    * id : integer <<PK>>
    --
    # comment_id : integer <<FK>>
    # user_id : integer <<FK>>
    created_at : timestamptz
  }

  entity "ratings" as ratings <<community>> {
    * id : integer <<PK>>
    --
    # recipe_id : integer <<FK>>
    # user_id : integer <<FK>>
    score : smallint
    created_at : timestamptz
    updated_at : timestamptz
  }

  entity "favorites" as favorites <<community>> {
    * id : integer <<PK>>
    --
    # recipe_id : integer <<FK>>
    # user_id : integer <<FK>>
    created_at : timestamptz
  }

  entity "follows" as follows <<community>> {
    * id : integer <<PK>>
    --
    # follower_id : integer <<FK>>
    # following_id : integer <<FK>>
    created_at : timestamptz
  }
}

' ---------------------------------------------------------------------

  entity "notifications" as notifications <<notify>> {
    * id : integer <<PK>>

    # user_id : integer <<FK>>
